// Cache-aside: hit Redis first, fall back to SQL Server, tenant-scoped key
[HttpGet("{id:guid}")]
public async Task<IActionResult> GetTrip(Guid id, CancellationToken ct)
{
    var key = $"{_tenant.Id}:trip:{id}";
    var hit = await _cache.GetStringAsync(key, ct);
    if (hit is not null)
        return Ok(JsonSerializer.Deserialize<TripDto>(hit));

    var trip = await _repo.GetByIdAsync(id, ct);
    if (trip is null) return NotFound();

    await _cache.SetStringAsync(key, JsonSerializer.Serialize(trip), new DistributedCacheEntryOptions
        { AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(5) }, ct);
    return Ok(trip);
}
